  full_name text,

  location text,

  total_qty integer,
  note text

DO $body$
DECLARE
  v_company uuid := 'e1d4a1f8-bab8-41b9-b8c8-de7e9a910b13';
  v_user uuid := '02e43f0e-ee95-40ef-8d10-111b963c6d94';
  v_variant_problems text;
  v_stock_problems text;
  v_dup_emails text;
  v_result json;
  r record;
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM public.profiles
    WHERE id = v_user AND company_id = v_company
  ) THEN
    RAISE EXCEPTION 'Profile % is not in company %', v_user, v_company;
  END IF;

  CREATE TEMP TABLE _leaders ON COMMIT DROP AS
  SELECT s.email, p.id AS leader_id, p.full_name, p.role
  FROM (SELECT DISTINCT lower(trim(email)) AS email FROM _hub_src) s
  LEFT JOIN public.profiles p
    ON p.company_id = v_company
   AND lower(trim(p.email)) = s.email;

  SELECT string_agg(email, E'\n' ORDER BY email)
  INTO v_dup_emails
  FROM (
    SELECT email
    FROM _leaders
    WHERE leader_id IS NOT NULL
    GROUP BY email
    HAVING COUNT(*) > 1
  ) d;

  IF v_dup_emails IS NOT NULL THEN
    RAISE EXCEPTION E'More than one profile in this company uses these emails:%\n%', E'\n', v_dup_emails;
  END IF;

  IF NOT EXISTS (SELECT 1 FROM _leaders WHERE leader_id IS NOT NULL) THEN
    RAISE EXCEPTION 'None of the hub emails match a profile in this company';
  END IF;

  CREATE TEMP TABLE _lines ON COMMIT DROP AS
  SELECT
    h.location,
    lower(trim(h.email)) AS email,
    l.leader_id,
    l.full_name,
    l.role,
    h.product AS src_brand,
    h.variant AS src_variant,
    COALESCE(bm.catalog_brand, h.product) AS brand,
    COALESCE(vm.catalog_variant, h.variant) AS variant,
    COALESCE(
      vm.catalog_type,
      CASE lower(trim(h.variant_type))
        WHEN 'device' THEN 'battery'
        ELSE lower(trim(h.variant_type))
      END
    ) AS variant_type,
    h.qty
  FROM _hub_src h
  JOIN _leaders l
    ON l.email = lower(trim(h.email))
   AND l.leader_id IS NOT NULL
  LEFT JOIN _brand_map bm
    ON upper(trim(bm.src_brand)) = upper(trim(h.product))
  LEFT JOIN _variant_map vm
    ON upper(trim(vm.src_brand)) = upper(trim(h.product))
   AND upper(trim(vm.src_variant)) = upper(trim(h.variant))
  WHERE h.qty > 0;

  CREATE TEMP TABLE _matched ON COMMIT DROP AS
  SELECT
    ln.location,
    ln.email,
    ln.leader_id,
    ln.full_name,
    ln.role,
    ln.src_brand,
    ln.src_variant,
    ln.brand,
    ln.variant,
    ln.variant_type,
    SUM(ln.qty)::int AS qty,
    COUNT(v.id) AS match_count,
    (ARRAY_AGG(v.id ORDER BY v.id) FILTER (WHERE v.id IS NOT NULL))[1] AS variant_id
  FROM _lines ln
  LEFT JOIN public.brands b
    ON b.company_id = v_company
   AND upper(trim(b.name)) = upper(trim(ln.brand))
  LEFT JOIN public.variants v
    ON v.company_id = v_company
   AND v.brand_id = b.id
   AND upper(trim(v.name)) = upper(trim(ln.variant))
   AND lower(trim(v.variant_type)) = ln.variant_type
  GROUP BY
    ln.location, ln.email, ln.leader_id, ln.full_name, ln.role,
    ln.src_brand, ln.src_variant, ln.brand, ln.variant, ln.variant_type;

  SELECT string_agg(
    format('%s | %s -> %s | %s | qty %s | %s',
      src_brand, src_variant, variant, variant_type, qty,
      CASE WHEN match_count = 0 THEN 'not found' ELSE 'more than one match' END),
    E'\n' ORDER BY src_brand, src_variant
  )
  INTO v_variant_problems
  FROM (
    SELECT DISTINCT src_brand, src_variant, variant, variant_type, qty, match_count
    FROM _matched
    WHERE match_count <> 1
  ) bad;

  IF v_variant_problems IS NOT NULL THEN
    RAISE EXCEPTION E'Fix these variants before allocating:%\n%', E'\n', v_variant_problems;
  END IF;

  CREATE TEMP TABLE _need ON COMMIT DROP AS
  SELECT variant_id, brand, variant, variant_type, SUM(qty)::int AS qty
  FROM _matched
  GROUP BY variant_id, brand, variant, variant_type;

  SELECT string_agg(
    format('%s | %s | %s | need %s | %s',
      brand, variant, variant_type, qty,
      CASE
        WHEN mi_count = 0 THEN 'not in main inventory'
        WHEN mi_count > 1 THEN 'more than one main inventory row'
        ELSE format('available %s', available)
      END),
    E'\n' ORDER BY brand, variant
  )
  INTO v_stock_problems
  FROM (
    SELECT
      n.brand,
      n.variant,
      n.variant_type,
      n.qty,
      COUNT(mi.id) AS mi_count,
      COALESCE(SUM(mi.stock - COALESCE(mi.allocated_stock, 0)), 0)::int AS available
    FROM _need n
    LEFT JOIN public.main_inventory mi
      ON mi.company_id = v_company
     AND mi.variant_id = n.variant_id
    GROUP BY n.brand, n.variant, n.variant_type, n.qty, n.variant_id
  ) s
  WHERE mi_count <> 1 OR available < qty;

  IF v_stock_problems IS NOT NULL THEN
    RAISE EXCEPTION E'Main inventory cannot cover these lines. Receive stock first, then run this again:%\n%', E'\n', v_stock_problems;
  END IF;

  INSERT INTO _alloc_log (email, note)
  SELECT email, 'email not in this company'
  FROM _leaders
  WHERE leader_id IS NULL;

  FOR r IN
    SELECT
      leader_id,
      email,
      full_name,
      role,
      MIN(location) AS location,
      COUNT(*)::int AS line_count,
      SUM(qty)::int AS total_qty,
      jsonb_agg(
        jsonb_build_object('variant_id', variant_id, 'quantity', qty)
        ORDER BY brand, variant
      ) AS items
    FROM _matched
    GROUP BY leader_id, email, full_name, role
  LOOP
    v_result := public.allocate_batch_to_leader(r.leader_id, r.items, v_user, NULL);
    IF COALESCE((v_result->>'success')::boolean, false) IS NOT TRUE THEN
      RAISE EXCEPTION '%', COALESCE(v_result->>'error', 'Allocation failed for ' || r.email);
    END IF;
    INSERT INTO _alloc_log (
      email, full_name, role, location, allocation_id, line_count, total_qty, note
    ) VALUES (
      r.email, r.full_name, r.role, r.location,
      (v_result->>'allocation_id')::uuid,
      r.line_count, r.total_qty, 'allocated'
    );
  END LOOP;
END
$body$;

COMMIT;

SELECT email, full_name, role, location, allocation_id, line_count, total_qty, note
FROM _alloc_log
ORDER BY note, email;
